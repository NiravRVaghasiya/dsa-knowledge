class Trie:
    def __init__(self):
        self.root = {}
    def insert(self, word):
        node = self.root
        for ch in word:
            node = node.setdefault(ch, {})
        node["\$"] = True          # end-of-word marker
    def starts_with(self, prefix):
        node = self.root
        for ch in prefix:
            if ch not in node:
                return False
            node = node[ch]
        return True

t = Trie()
for w in ["cat", "car", "card", "dog"]:
    t.insert(w)
print(t.starts_with("ca"))   # -> True
print(t.starts_with("dax"))  # -> False
`}
