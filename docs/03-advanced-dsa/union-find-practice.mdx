  code={`# Union-Find (Disjoint Set Union) with path compression + union by size.
class DSU:
    def __init__(self, n):
        self.parent = list(range(n))
        self.size = [1] * n
    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]  # path compression
            x = self.parent[x]
        return x
    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False           # already connected
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        return True

d = DSU(6)
for a, b in [(0,1),(2,3),(1,3),(4,5)]:
    d.union(a, b)
print(d.find(0) == d.find(3))   # -> True  (0-1-3-2 connected)
print(d.find(0) == d.find(4))   # -> False
`}
