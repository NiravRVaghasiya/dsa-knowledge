# Dijkstra: shortest distances from source in a weighted graph (non-negative).
def dijkstra(graph, src):
    dist = {u: float("inf") for u in graph}
    dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if d + w < dist[v]:
                dist[v] = d + w
                heapq.heappush(pq, (dist[v], v))
    return dist

g = {"A":[("B",1),("C",4)], "B":[("C",2),("D",5)], "C":[("D",1)], "D":[]}
print(dijkstra(g, "A"))   # -> {'A':0,'B':1,'C':3,'D':4}
`}
