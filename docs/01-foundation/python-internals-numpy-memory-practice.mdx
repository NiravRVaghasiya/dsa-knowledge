a = [1, 2, 3]
b = a            # alias — same object
c = a[:]         # shallow copy — new object
b.append(4)
print("a:", a, "| b is a:", b is a)   # b mutated a
print("c:", c, "| c is a:", c is a)   # c untouched

# Small-int caching
x = 256; y = 256
print("256 cached:", x is y)          # True
x = 257; y = 257
print("257 cached:", x is y)          # often False
