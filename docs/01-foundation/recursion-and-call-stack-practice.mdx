  code={`# Fibonacci: naive recursion vs memoized — watch the call count.
calls = 0
def fib_naive(n):
    global calls; calls += 1
    return n if n < 2 else fib_naive(n-1) + fib_naive(n-2)

from functools import lru_cache
@lru_cache(None)
def fib_fast(n):
    return n if n < 2 else fib_fast(n-1) + fib_fast(n-2)

print("fib_naive(20) =", fib_naive(20), "in", calls, "calls")
print("fib_fast(20)  =", fib_fast(20), "(linear calls)")
`}
