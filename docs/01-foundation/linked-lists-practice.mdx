## 🐍 Your turn

<PythonRunner
  title="🐍 Linked Lists — Practice"
  height={380}
  code={`# Reverse a singly linked list. Node = {"v": value, "next": node_or_None}.
def reverse(head):
    prev = None
    while head:
        nxt = head["next"]   # save next
        head["next"] = prev  # re-point backward
        prev = head          # advance prev
        head = nxt           # advance head
    return prev              # new head

# build 1->2->3, reverse, print
head = {"v": 1, "next": {"v": 2, "next": {"v": 3, "next": None}}}
r = reverse(head)
out = []
while r:
    out.append(r["v"]); r = r["next"]
print(out)   # -> [3, 2, 1]
`}
/>
