  code={`# Rotate an array to the right by k steps, in-place, O(1) extra space.
def rotate(nums, k):
    n = len(nums)
    k %= n
    nums[:] = nums[::-1]          # reverse all
    nums[:k] = nums[:k][::-1]     # reverse first k
    nums[k:] = nums[k:][::-1]     # reverse the rest
    return nums

print(rotate([1,2,3,4,5,6,7], 3))   # -> [5,6,7,1,2,3,4]
`}
