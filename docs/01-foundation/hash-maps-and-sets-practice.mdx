  title="🐍 Hash Maps & Sets — Practice"

  code={`# Two Sum (unsorted) in O(n) using a hash map of value -> index.
def two_sum(nums, target):
    seen = {}
    for i, x in enumerate(nums):
        if target - x in seen:
            return [seen[target - x], i]
        seen[x] = i
    return []

print(two_sum([2, 7, 11, 15], 9))   # -> [0, 1]
print(two_sum([3, 2, 4], 6))        # -> [1, 2]
`}
