  code={`# Classic iterative binary search. Return the index of target, or -1.
def binary_search(a, target):
    lo, hi = 0, len(a) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if a[mid] == target:
            return mid
        elif a[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1

print(binary_search([1,3,5,7,9,11,13,15], 11))  # -> 5
print(binary_search([1,3,5,7,9,11,13,15], 4))   # -> -1
`}
