# BFS traversal — return nodes in the order they are first visited.
def bfs(graph, start):
    seen = {start}
    q = deque([start])
    order = []
    while q:
        node = q.popleft()
        order.append(node)
        for nb in graph[node]:
            if nb not in seen:
                seen.add(nb)
                q.append(nb)
    return order

g = {1: [2, 3], 2: [4, 5], 3: [6], 4: [], 5: [6], 6: []}
print(bfs(g, 1))   # -> [1, 2, 3, 4, 5, 6]
`}
