<PythonRunner
  title="🐍 Stacks & Queues — Practice"
  height={380}
  code={`# Valid parentheses using a stack. Return True if every bracket is matched.
def is_valid(s):
    stack = []
    pairs = {')': '(', ']': '[', '}': '{'}
    for c in s:
        if c in '([{':
            stack.append(c)
        else:
            if not stack or stack.pop() != pairs[c]:
                return False
    return not stack

print(is_valid("([{}])"))   # -> True
print(is_valid("(]"))       # -> False
`}
