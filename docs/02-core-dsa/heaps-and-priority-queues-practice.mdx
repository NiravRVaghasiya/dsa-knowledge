# Use a heap to find the k smallest elements in O(n log k).
def k_smallest(nums, k):
    return heapq.nsmallest(k, nums)

# ...and a running "k largest" with a min-heap of size k:
def k_largest(nums, k):
    h = []
    for x in nums:
        heapq.heappush(h, x)
        if len(h) > k:
            heapq.heappop(h)      # evict smallest -> keeps k largest
    return sorted(h, reverse=True)

print(k_smallest([5, 2, 8, 1, 4, 9], 3))  # -> [1, 2, 4]
print(k_largest([5, 2, 8, 1, 4, 9], 3))   # -> [9, 8, 5]
`}
