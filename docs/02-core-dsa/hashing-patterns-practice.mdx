  code={`# Count subarrays whose sum equals k, using prefix-sum counts.
from collections import defaultdict
def subarray_sum(nums, k):
    count = 0
    prefix = 0
    seen = defaultdict(int)
    seen[0] = 1
    for x in nums:
        prefix += x
        count += seen[prefix - k]   # a prior prefix makes a window summing to k
        seen[prefix] += 1
    return count

print(subarray_sum([1, 1, 1], 2))       # -> 2
print(subarray_sum([1, 2, 3], 3))       # -> 2
`}
