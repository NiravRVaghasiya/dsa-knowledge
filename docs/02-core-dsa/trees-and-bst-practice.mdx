<PythonRunner
  title="🐍 Trees & BST — Practice"
  height={380}
  code={`# In-order traversal of a binary tree. Returns values Left -> Root -> Right.
# A node is a dict: {"v": value, "l": left_or_None, "r": right_or_None}
def inorder(node):
    if node is None:
        return []
    return inorder(node["l"]) + [node["v"]] + inorder(node["r"])

tree = {"v": 5,
        "l": {"v": 3, "l": {"v": 2, "l": None, "r": None},
                       "r": {"v": 4, "l": None, "r": None}},
        "r": {"v": 8, "l": {"v": 7, "l": None, "r": None},
                       "r": {"v": 9, "l": None, "r": None}}}
print(inorder(tree))   # -> [2, 3, 4, 5, 7, 8, 9]  (sorted, because it is a BST)
`}
