# Topological sort of an agent DAG (Kahn's algorithm).
def topo_sort(graph):
    indeg = {u: 0 for u in graph}
    for u in graph:
        for v in graph[u]:
            indeg[v] += 1
    q = deque([u for u in graph if indeg[u] == 0])
    order = []
    while q:
        u = q.popleft(); order.append(u)
        for v in graph[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return order if len(order) == len(graph) else None   # None if cycle

dag = {"ingest":["embed"], "embed":["retrieve"], "retrieve":["generate"], "generate":[]}
print(topo_sort(dag))   # -> ['ingest','embed','retrieve','generate']
`}
