  code={`# Nearest neighbor by brute force vs. a KD-tree (via scipy if available).
import math
def nearest(points, query):
    best, bd = None, float("inf")
    for p in points:
        d = math.dist(p, query)
        if d < bd:
            bd, best = d, p
    return best, round(bd, 3)

pts = [(1,2),(3,4),(5,1),(0,0),(4,4)]
print(nearest(pts, (4, 3)))   # -> ((4,4), 1.0)
`}
