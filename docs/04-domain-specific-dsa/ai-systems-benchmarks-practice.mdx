## 🐍 Benchmark A — ANN: recall vs. distance computations

<PythonRunner
  title="🐍 Brute force vs. simplified IVF — recall@k and work"
  height={640}
  code={`import numpy as np
np.random.seed(0)

# Synthetic dataset: n vectors in d dims.
n, d, k = 2000, 32, 10
X = np.random.randn(n, d).astype(np.float32)
queries = np.random.randn(50, d).astype(np.float32)

def brute_force(q):
    d2 = ((X - q) ** 2).sum(axis=1)     # n distance computations
    return np.argsort(d2)[:k], n

# Simplified IVF: cluster into nlist buckets, probe nprobe nearest centroids.
nlist = 40
# quick k-means-ish: random centroids + one Lloyd step
cent = X[np.random.choice(n, nlist, replace=False)].copy()
assign = np.argmin(((X[:, None, :] - cent[None, :, :]) ** 2).sum(2), axis=1)
for c in range(nlist):
    pts = X[assign == c]
    if len(pts): cent[c] = pts.mean(0)
assign = np.argmin(((X[:, None, :] - cent[None, :, :]) ** 2).sum(2), axis=1)
buckets = {c: np.where(assign == c)[0] for c in range(nlist)}

def ivf_search(q, nprobe):
    cd = ((cent - q) ** 2).sum(1)
    probe = np.argsort(cd)[:nprobe]                 # nearest centroids
    cand = np.concatenate([buckets[c] for c in probe]) if len(probe) else np.array([], int)
    work = nlist + len(cand)                         # centroid + candidate distances
    if len(cand) == 0: return np.array([], int), work
    d2 = ((X[cand] - q) ** 2).sum(1)
    return cand[np.argsort(d2)[:k]], work

# Ground truth
truth = [set(brute_force(q)[0]) for q in queries]

print(f"dataset: n={n}, d={d}, k={k}, nlist={nlist}")
print(f"{'nprobe':>7} | {'recall@k':>9} | {'avg dist comps':>14} | {'vs brute':>9}")
print("-"*50)
for nprobe in [1, 2, 4, 8, 16, nlist]:
    recalls, works = [], []
    for q, t in zip(queries, truth):
        got, w = ivf_search(q, nprobe)
        recalls.append(len(set(got) & t) / k)
        works.append(w)
    r, w = np.mean(recalls), np.mean(works)
    print(f"{nprobe:>7} | {r:>9.3f} | {w:>14.0f} | {n/w:>8.1f}x")
print(f"\\nbrute force: recall=1.000, dist comps={n} (the ground truth)")
print("Observe: more probes -> higher recall AND more work. The ANN dial.")
`}

## 🐍 Benchmark B — Attention: the two-regime cost

<PythonRunner
  title="🐍 Score-matrix growth vs. projection cost as L grows"
  height={560}
  code={`import numpy as np, time
np.random.seed(0)

D = 256            # model dim (fixed); H, B held at 1 for clarity
print(f"model dim D={D}")
print(f"{'L':>6} | {'score L^2':>12} | {'proj L*D^2':>14} | {'QK^T ms':>9} | regime")
print("-"*62)
prev_t = None
for L in [64, 128, 256, 512, 1024]:
    Q = np.random.randn(L, D).astype(np.float32)
    K = np.random.randn(L, D).astype(np.float32)
    # time the QK^T score computation (median of a few runs)
    ts = []
    for _ in range(3):
        t0 = time.perf_counter()
        S = Q @ K.T                       # (L, L) score matrix
        ts.append(time.perf_counter() - t0)
    ms = sorted(ts)[1] * 1000
    score = L * L                          # exact: attention-matrix entries
    proj = L * D * D                       # Q/K/V/O projection work
    regime = "projection-bound" if proj > score * D else "attention-bound"
    ratio = f"({ms/prev_t:.1f}x prev)" if prev_t else ""
    print(f"{L:>6} | {score:>12,} | {proj:>14,} | {ms:>7.2f} {ratio}")
    prev_t = ms
print("\\nScore entries quadruple when L doubles (L^2). The projection term")
print("(L*D^2) dominates until L approaches D -- 'O(n^2)' alone hides this.")
`}

## 🐍 Benchmark C — LRU cache hit/miss vs. access pattern

<PythonRunner
  title="🐍 LRU hit rate under locality vs. scan workloads"
  height={600}
  code={`from collections import OrderedDict
import random
random.seed(0)

class LRU:
    def __init__(self, cap):
        self.cap = cap; self.d = OrderedDict(); self.hits = self.miss = 0
    def get(self, key):
        if key in self.d:
            self.d.move_to_end(key); self.hits += 1; return True
        self.miss += 1
        self.d[key] = 1
        if len(self.d) > self.cap: self.d.popitem(last=False)
        return False
    def rate(self):
        tot = self.hits + self.miss
        return self.hits / tot if tot else 0.0

N = 10000
cap = 100

# 1) temporal locality: Zipf-like — a few hot keys dominate
def zipf_keys(n, m=500, s=1.3):
    weights = [1.0/(i**s) for i in range(1, m+1)]
    tot = sum(weights); probs = [w/tot for w in weights]
    return random.choices(range(m), weights=probs, k=n)

# 2) scan: each of many distinct keys touched once (more than capacity)
scan_keys = list(range(N))

for name, keys in [("locality (Zipf)", zipf_keys(N)), ("scan (all distinct)", scan_keys)]:
    c = LRU(cap)
    for k in keys: c.get(k)
    print(f"{name:>22}: hit rate = {c.rate():.3f}")

print("\\ncapacity sweep on the locality workload:")
keys = zipf_keys(N)
for cap in [10, 50, 100, 250, 500]:
    c = LRU(cap)
    for k in keys: c.get(k)
    print(f"  capacity {cap:>4}: hit rate = {c.rate():.3f}")
print("\\nLRU shines under locality, collapses on a scan (scan-resistance failure).")
print("This is exactly KV/prefix-cache behavior: value = f(access-pattern locality).")
`}

## 🧠 Challenge yourself

1. In Benchmark A, add **IVF-PQ**: quantize each vector to a few bytes and measure the recall drop vs. the memory saved. Then add a **rerank** step (re-score the shortlist with full-precision distances) and show recall recovers.
2. In Benchmark B, add the **`A·V`** step and confirm total attention time follows `a·L·D² + b·L²·D`; fit `a, b` and find the crossover `L`.
3. In Benchmark C, replace LRU with a **scan-resistant** policy (e.g. 2Q / segmented LRU) and show it survives the scan workload where plain LRU fails.
4. Build a tiny **recall-vs-QPS Pareto plot** for Benchmark A by pairing each `nprobe`'s recall with `1/work` as a QPS proxy.
