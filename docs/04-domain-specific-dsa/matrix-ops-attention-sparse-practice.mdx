import numpy as np
def attention(Q, K, V):
    d = Q.shape[-1]
    scores = Q @ K.T / np.sqrt(d)
    scores -= scores.max(axis=-1, keepdims=True)   # numerical stability
    w = np.exp(scores)
    w /= w.sum(axis=-1, keepdims=True)
    return w @ V

Q = np.array([[1.0, 0.0]]); K = np.array([[1.0,0.0],[0.0,1.0]]); V = np.array([[10.0],[20.0]])
print(attention(Q, K, V))   # weighted toward the matching key
`}
