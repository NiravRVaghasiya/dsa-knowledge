# 🎮 ANN Search (Practice)

  code={`# Brute-force cosine similarity search (the baseline ANN methods approximate).
import math
def cosine(a, b):
    dot = sum(x*y for x, y in zip(a, b))
    na = math.sqrt(sum(x*x for x in a))
    nb = math.sqrt(sum(y*y for y in b))
    return dot / (na * nb)

def top_k(query, vectors, k=2):
    scored = [(cosine(query, v), i) for i, v in enumerate(vectors)]
    scored.sort(reverse=True)
    return scored[:k]

db = [[1,0,0],[0.9,0.1,0],[0,1,0],[0,0,1]]
print(top_k([1,0,0], db, 2))   # nearest to [1,0,0]
`}
