from collections import Counter
def most_frequent_pair(tokens):
    pairs = Counter(zip(tokens, tokens[1:]))
    return pairs.most_common(1)[0][0] if pairs else None

def merge(tokens, pair):
    out, i = [], 0
    while i < len(tokens):
        if i < len(tokens)-1 and (tokens[i], tokens[i+1]) == pair:
            out.append(tokens[i] + tokens[i+1]); i += 2
        else:
            out.append(tokens[i]); i += 1
    return out

toks = list("low low low lower")
for _ in range(3):
    p = most_frequent_pair(toks)
    toks = merge(toks, p)
print(toks)
`}
