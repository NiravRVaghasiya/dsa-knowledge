def pagerank(graph, d=0.85, iters=20):
    nodes = list(graph)
    n = len(nodes)
    pr = {u: 1/n for u in nodes}
    for _ in range(iters):
        new = {}
        for u in nodes:
            s = sum(pr[v] / len(graph[v]) for v in nodes if u in graph[v])
            new[u] = (1 - d) / n + d * s
        pr = new
    return {u: round(v, 3) for u, v in pr.items()}

g = {"A":["B","C"], "B":["C"], "C":["A"], "D":["C"]}
print(pagerank(g))   # C ranks highest (most incoming)
`}
