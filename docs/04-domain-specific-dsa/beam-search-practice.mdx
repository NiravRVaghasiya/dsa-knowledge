  code={`# Beam search: keep the top-b partial sequences by cumulative log-prob.
import math
def beam_search(step_probs, beam=2):
    # step_probs[t] = dict(token -> prob) at step t (independent toy model)
    beams = [([], 0.0)]   # (sequence, logprob)
    for probs in step_probs:
        cand = []
        for seq, lp in beams:
            for tok, p in probs.items():
                cand.append((seq + [tok], lp + math.log(p)))
        cand.sort(key=lambda x: x[1], reverse=True)
        beams = cand[:beam]
    return beams[0]

steps = [{"a":0.6,"b":0.4}, {"x":0.7,"y":0.3}, {"m":0.9,"n":0.1}]
seq, lp = beam_search(steps, beam=2)
print(seq, round(lp, 3))   # -> ['a','x','m'] (highest-prob path)
`}
